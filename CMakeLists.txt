cmake_minimum_required(VERSION 3.25)
project(ai_chess VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHESS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(chess_core cpp/board.cpp cpp/search.cpp cpp/book.cpp)
target_include_directories(chess_core PUBLIC cpp)
target_compile_options(chess_core PRIVATE -Wall -Wextra -Wpedantic)
if(CHESS_SANITIZE)
  target_compile_options(chess_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(chess_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(ai_chess_engine cpp/main.cpp)
target_link_libraries(ai_chess_engine PRIVATE chess_core Threads::Threads)
add_executable(chess_book cpp/book_main.cpp)
target_link_libraries(chess_book PRIVATE chess_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE chess_core)
  add_test(NAME core COMMAND core_tests)
endif()
